# Purpose:
#   Unified environment for the JP01 Voice LLM Demo integrating:

# Virtual Environment Setup (macOS)
# -----------------------------------------------------------------------------
#   # Install pyenv (if not yet installed)
#   brew install pyenv portaudio ffmpeg libsndfile
#
#   # Create project folder
#   mkdir -p ~/dev/voice-llm-demo && cd ~/dev/voice-llm-demo
#
#   # Set Python version (3.13.x recommended)
#   pyenv install 3.12.6 -s
#   pyenv local 3.12.6
#
#   # Create and activate a virtual environment
#   python -m venv .venv
#   source .venv/bin/activate
#
#   # Upgrade pip/wheel/setuptools
#   python -m pip install --upgrade pip wheel setuptools
#
# -----------------------------------------------------------------------------
# Homebrew Prerequisites (macOS)
# -----------------------------------------------------------------------------
#   # Required for sounddevice and audio support
#   brew install portaudio libsndfile ffmpeg
#
#   # Optional (useful for testing and debugging audio/TTS)

#
#   # Verify installation
#   brew list portaudio libsndfile ffmpeg
